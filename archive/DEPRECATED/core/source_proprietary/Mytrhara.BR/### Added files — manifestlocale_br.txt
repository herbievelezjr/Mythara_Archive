### Unit test file — core/tests/test_locale_braille.py
```python
import json
import pathlib
from mythara.br.encoder import BrailleEncoder
from mythara.br.config import BrConfig

ROOT = pathlib.Path(__file__).resolve().parent
FIXTURES = ROOT / "locale_fixtures"

def load_fixture(name):
    return json.load(open(FIXTURES / name, "r", encoding="utf-8"))

def test_spanish_locale_examples():
    fx = load_fixture("sample_es_inputs.json")
    cfg = BrConfig()
    cfg.transliteration_policy = "latinize_then_encode"
    encoder = BrailleEncoder(cfg, adapter_map=None)  # uses registered adapters
    for ex in fx["examples"]:
        res = encoder.encode(ex["text"], locale=fx["locale"])
        assert res.braille == ex["expected_braille"], f"es example failed: {ex['text']} -> {res.braille}"

def test_french_locale_examples():
    fx = load_fixture("sample_fr_inputs.json")
    cfg = BrConfig()
    cfg.transliteration_policy = "latinize_then_encode"
    encoder = BrailleEncoder(cfg, adapter_map=None)
    for ex in fx["examples"]:
        res = encoder.encode(ex["text"], locale=fx["locale"])
        assert res.braille == ex["expected_braille"], f"fr example failed: {ex['text']} -> {res.braille}"
```
