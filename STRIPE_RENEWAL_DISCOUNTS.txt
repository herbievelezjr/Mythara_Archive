```python
import stripe

def apply_renewal_discount(subscription_id, product_tier):
    coupon_codes = {
        "startup": "LOYALTY_STARTUP_10",
        "corporate": "LOYALTY_CORPORATE_8",
        "midmarket": "LOYALTY_MIDMARKET_6",
        "enterprise": "LOYALTY_ENTERPRISE_4",
        "global": "LOYALTY_GLOBAL_2"
    }
    
    coupon = coupon_codes.get(product_tier.lower())
    
    if coupon:
        stripe.Subscription.modify(
            subscription_id,
            coupon=coupon
        )
```
